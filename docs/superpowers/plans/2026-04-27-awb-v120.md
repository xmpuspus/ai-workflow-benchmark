# AWB v1.2.0 Implementation Plan

> **For agentic workers:** This is a release plan. Each task is one logical commit. Execute in order; do NOT batch commits. Run `pytest tests/ -x -q` and `ruff check awb/ tests/` after every commit. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Fix 7 P0 trust blockers identified in the v2 strategy, then ship a scoped P1 differentiation slice (task-set hash, fresh/verified metadata, OTel-aligned trace artifact, `awb trace grade`, Production Readiness Score) without expanding scope.

**Architecture:** P0 = surgical fixes to existing modules, no API breakage. P1 = three new subsystems with clear seams: hashing (single function in existing `awb/scoring/integrity.py`), trace artifact (new package `awb/trace/` mapped to OpenTelemetry GenAI conventions for LLM/tool spans, custom span names for shell/file/test), and readiness scoring (new `awb/scoring/readiness.py` composing existing dimensions plus 2 trace-derived ones). Result schema bumps from v1 to v2 — strict, versioned, with `task_set_hash` and optional `trace_path`. `awb migrate-results` extended to handle v1→v2.

**Tech Stack:** Python 3.11+, dataclasses (no Pydantic — preserve minimal-deps design), jsonschema validator, click CLI, hashlib for SHA-256, json for JSONL persistence. Match existing code style (100-char line length, ruff lint).

**Trace schema source:** OTel GenAI semantic conventions where applicable (`gen_ai.system`, `gen_ai.request.model`, `gen_ai.usage.input_tokens`, `gen_ai.usage.output_tokens`, `gen_ai.tool.name`, `gen_ai.tool.call_id`). Custom span names for AWB-specific events (`task.shell_command`, `task.file_edit`, `task.test_run`). JSONL persistence: one event per line under `<run_dir>/<task_id>_<tool>.trace.jsonl`.

```python
async def _run_command(cmd: str, workspace: Path) -> tuple[int, str, str]:
    try:
        proc = await asyncio.create_subprocess_shell(
            cmd, cwd=workspace,
            stdout=asyncio.subprocess.PIPE, stderr=asyncio.subprocess.PIPE,
        )
        try:
            stdout, stderr = await asyncio.wait_for(proc.communicate(), timeout=180)
        except TimeoutError:
            proc.kill()
            await proc.communicate()
            return 1, "", "[TIMEOUT]"
        return proc.returncode, stdout.decode(errors="replace"), stderr.decode(errors="replace")
    except FileNotFoundError as exc:
        return 127, "", f"[scanner not found] {exc}"
```

```python
def classify_failure(result: RunResult, task: TaskDefinition) -> str:
    if result.outcome.success:
        return "success"
    if result.metrics.wall_clock_seconds >= task.constraints.timeout_seconds * 0.95:
        return "timeout"
    if result.quality.test_regressions > 0:
        return "regression_introduced"
    if result.metrics.files_modified == 0:
        return "no_edits_made"
    has_test_criteria = any(
        "test" in c.criterion.lower() or "pass" in c.criterion.lower()
        for c in result.outcome.breakdown
    )
    test_criteria_failed = any(
        ("test" in c.criterion.lower() or "pass" in c.criterion.lower()) and not c.passed
        for c in result.outcome.breakdown
    )
    if has_test_criteria and test_criteria_failed:
        return "test_error"
    if result.outcome.partial_credit_score > 0:
        return "partial_completion"
    return "code_error"
```

```python
import hashlib
from pathlib import Path


def compute_task_set_hash(tasks_dir: Path | str) -> str:
    """Return SHA-256 over canonicalized concatenation of all task YAML files.

    Walk the dir recursively, sort paths alphabetically, hash file bytes
    in that order. Produces a stable identifier for "this exact task set"
    that callers can embed in result records and leaderboard rows.
    """
    root = Path(str(tasks_dir))
    h = hashlib.sha256()
    paths = sorted(p for p in root.glob("**/*.yaml") if not p.name.startswith("_"))
    for p in paths:
        rel = p.relative_to(root).as_posix().encode()
        h.update(b"\x00path:")
        h.update(rel)
        h.update(b"\x00bytes:")
        h.update(p.read_bytes())
    return h.hexdigest()
```

```python
"""Append-only JSONL writer for trace events."""
from __future__ import annotations

import json
from pathlib import Path


class TraceWriter:
    def __init__(self, path: Path) -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._fh = None

    def write(self, span: dict) -> None:
        if self._fh is None:
            self._fh = open(self.path, "a")
        self._fh.write(json.dumps(span) + "\n")
        self._fh.flush()

    def close(self) -> None:
        if self._fh is not None:
            self._fh.close()
            self._fh = None

    def __enter__(self):
        return self

    def __exit__(self, *exc):
        self.close()


def load_trace(path: Path) -> list[dict]:
    p = Path(path)
    if not p.exists():
        return []
    out = []
    with p.open() as f:
        for line in f:
            line = line.strip()
            if line:
                out.append(json.loads(line))
    return out
```

```python
"""Grade a trace JSONL by 4 behavior dimensions, each 0-100."""
from __future__ import annotations

from collections.abc import Iterable
from pathlib import Path

from awb.trace import FILE_EDIT, SHELL_COMMAND, TEST_RUN, TOOL_USE
from awb.trace.jsonl import load_trace


def _is_test_path(path: str) -> bool:
    parts = (path or "").lower().replace("\\", "/").split("/")
    return any(p.startswith("test") for p in parts) or "tests" in parts


def _read_attr(span: dict, key: str, default=""):
    return (span.get("attributes") or {}).get(key, default)


def grade_trace(path: Path, files_to_examine: list[str] | None = None) -> dict[str, int]:
    spans = load_trace(path)
    return {
        "read_tests_before_edit": _grade_read_tests_before_edit(spans),
        "ran_verification_after_change": _grade_ran_verification_after_change(spans),
        "no_out_of_scope_edits": _grade_no_out_of_scope_edits(spans, files_to_examine or []),
        "no_repeated_failing_command_loop": _grade_no_repeated_failing_loop(spans),
    }


def _grade_read_tests_before_edit(spans: Iterable[dict]) -> int:
    saw_test_read = False
    for s in spans:
        name = s.get("span_name")
        if name == TOOL_USE:
            tname = (_read_attr(s, "gen_ai.tool.name") or "").lower()
            fp = _read_attr(s, "file.path", "")
            if tname in {"read", "view", "open"} and _is_test_path(fp):
                saw_test_read = True
        elif name == FILE_EDIT:
            return 100 if saw_test_read else 0
    return 100  # no edits at all -> trivially passes


def _grade_ran_verification_after_change(spans: Iterable[dict]) -> int:
    last_edit_idx = -1
    last_test_idx = -1
    for i, s in enumerate(spans):
        if s.get("span_name") == FILE_EDIT:
            last_edit_idx = i
        elif s.get("span_name") in (TEST_RUN, SHELL_COMMAND):
            cmd = (_read_attr(s, "shell.command") or "").lower()
            if s.get("span_name") == TEST_RUN or "pytest" in cmd or "test" in cmd or "npm test" in cmd:
                last_test_idx = i
    if last_edit_idx < 0:
        return 100
    return 100 if last_test_idx > last_edit_idx else 0


def _grade_no_out_of_scope_edits(spans: Iterable[dict], files_to_examine: list[str]) -> int:
    if not files_to_examine:
        return 100
    allowed = set(files_to_examine)
    edited = []
    for s in spans:
        if s.get("span_name") == FILE_EDIT:
            edited.append(_read_attr(s, "file.path", ""))
    if not edited:
        return 100
    out_of_scope = sum(1 for e in edited if e and e not in allowed)
    return max(0, round(100 * (1 - out_of_scope / len(edited))))


def _grade_no_repeated_failing_loop(spans: Iterable[dict]) -> int:
    """Penalize when the same failing command runs 3+ times consecutively."""
    last_cmd = None
    streak = 0
    worst = 0
    for s in spans:
        if s.get("span_name") != SHELL_COMMAND:
            continue
        exit_code = _read_attr(s, "shell.exit_code", 0)
        cmd = _read_attr(s, "shell.command", "")
        if exit_code != 0 and cmd == last_cmd:
            streak += 1
            worst = max(worst, streak)
        else:
            streak = 0
        last_cmd = cmd
    if worst < 2:
        return 100
    if worst < 3:
        return 70
    return max(0, 50 - (worst - 3) * 15)
```

```python
"""Production Readiness Score — composite over 7 dimensions.

Designed to answer: "can this AI workflow safely ship real software?"
"""
from __future__ import annotations

# Weights sum to 1.0. Correctness dominates, then regression safety + security.
READINESS_DIMENSIONS: list[tuple[str, float]] = [
    ("correctness",        0.35),
    ("regression_safety",  0.20),
    ("security",           0.15),
    ("review_burden",      0.10),
    ("maintainability",    0.08),
    ("cost",               0.07),
    ("speed",              0.05),
]


def compute_readiness_score(
    *,
    correctness: float,
    regression_safety: float,
    security: float,
    review_burden: float,
    maintainability: float,
    cost: float,
    speed: float,
) -> float:
    inputs = {
        "correctness": correctness,
        "regression_safety": regression_safety,
        "security": security,
        "review_burden": review_burden,
        "maintainability": maintainability,
        "cost": cost,
        "speed": speed,
    }
    total = sum(inputs[name] * w for name, w in READINESS_DIMENSIONS)
    return round(total, 1)
```
